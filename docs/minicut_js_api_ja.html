1. はじめに & JavaScript アーキテクチャ

minicut は完全な JavaScript エンジン（Duktape 2.7.0）を内蔵しており、標準で完全な UTF-8 / Unicode 文字列操作をサポートしています。日本語の文字化けやバイト境界の問題がなく、国際化されたツールや自動化スクリプトをスムーズに開発できます。

スクリプトディレクトリの配置

ディレクトリ役割 / 実行タイミング概要
~/.config/minicut/autorun/ 起動時自動登録 起動時に自動実行され、メニューバー下の「ドック (Dock)」にアイコンボタンを登録します（.js ファイル対応）。
~/.config/minicut/scripts/ メニュークリック時実行 メインメニューバーの Scripts (スクリプト) メニューに登録され、クリック時にパネルを開いて実行します。
~/.config/minicut/icons/ ドック用アイコン画像 スクリプト名と同名の 32x32 PNG アイコン画像（例：text_animator.js 用の text_animator.png）。

2. JavaScript プラグインの基本構造

// ============================================================================
// JavaScript プラグインの基本例: custom_animator.js
// ============================================================================

var targetTrack = 0;
var textContent = "日本語テキスト字幕";

function applyAnimation() {
    minicut.push_undo();
    var clipId = timeline.place_text_clip(targetTrack, textContent, timeline.get_playhead(), 5.0);
    if (clipId) {
        clip.set_position(clipId, 0.0, 150.0);
        clip.set_scale(clipId, 1.2);
        timeline.add_keyframe(clipId, "opacity", 0.0, 0.0);
        timeline.add_keyframe(clipId, "opacity", 0.5, 1.0);
        minicut.set_status("字幕クリップを配置しました: " + textContent);
    }
}

plugin.init_panel = function() {
    gui.text("JavaScript テキストアニメーションツール");
    gui.separator();
    targetTrack = gui.slider_int("対象トラック", targetTrack, 0, 4);
    textContent = gui.input_text("字幕テキスト", textContent);
    
    if (gui.button("アニメーションテキストを配置", applyAnimation)) {
        // インライン処理
    }
};
