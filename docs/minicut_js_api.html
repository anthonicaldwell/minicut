1. Introduction & JavaScript Architecture

minicut embeds a complete JavaScript engine (Duktape 2.7.0) with native UTF-8 and Unicode string support. JavaScript plugins provide complete parity with the minicut scripting API, making it easy to create internationalized tools across all supported languages.

Script Directory Locations

LocationTrigger / RoleDescription
~/.config/minicut/autorun/ Startup / Auto-register .js files execute on startup and generate an icon button on the horizontal Tool Shelf Dock below the main menu bar.
~/.config/minicut/scripts/ Menu Click / On Demand .js files appear in the top-level Scripts menu on the main menu bar.
~/.config/minicut/icons/ Dock Icon Asset PNG icon images matching script stems (e.g., quick_fade_js.png for quick_fade_js.js).

2. JavaScript Plugin Architecture

// ==============================================================================
// Example JavaScript Plugin: sample_tool.js
// ==============================================================================

var targetTrack = 0;
var clipDuration = 4.0;
var mediaPath = "test_sample.png";

function placeMedia() {
    minicut.push_undo();
    var playhead = timeline.get_playhead();
    var cid = timeline.place_clip(mediaPath, targetTrack, playhead, clipDuration);
    if (cid && cid.length > 0) {
        clip.set_scale(cid, 1.25);
        timeline.add_keyframe(cid, "opacity", 0.0, 0.0);
        timeline.add_keyframe(cid, "opacity", 1.0, 1.0);
        minicut.set_status("Placed media clip: " + cid);
    }
}

function browseFile() {
    gui.file_dialog("Select Media", function(selected) {
        if (selected && selected.length > 0) {
            mediaPath = selected;
            minicut.set_status("Selected: " + selected);
        }
    });
}

plugin.init_panel = function() {
    gui.text_colored(0.24, 0.68, 0.91, 1.0, "Media Placer (JS)");
    gui.separator();
    targetTrack = gui.slider_int("Track Lane", targetTrack, 0, 4);
    clipDuration = gui.slider_float("Duration", clipDuration, 0.5, 30.0);
    gui.button("Browse Media...", browseFile);
    gui.button("Place on Timeline", placeMedia);
};

3.3. Timeline & Track Management (timeline)

FunctionReturnsDescription
timeline.get_playhead()numberGets playhead time in seconds.
timeline.set_playhead(t)voidSets playhead time in seconds.
timeline.place_clip(path, track, t, dur)stringPlaces media clip on track and returns clip ID.
timeline.add_keyframe(id, prop, t, val)voidAdds keyframe for property at timestamp.
timeline.clear_keyframes(id, prop)voidClears all keyframes for property.
timeline.select_clips(ids_array)voidSelects multiple clips simultaneously.

timeline.move_track_down(idx)voidMoves track down in stack.
timeline.set_track_name(idx, name)voidRenames track at index.
