1. Introduction & JavaScript Architecture

minicut embeds a complete JavaScript engine (Duktape 2.7.0) with native UTF-8 and Unicode string support. JavaScript plugins provide full parity with the minicut scripting API, making it easy to create internationalized tools in Japanese, Arabic, and other languages.

Script Directory Locations

LocationTrigger / RoleDescription
~/.config/minicut/autorun/ Startup / Auto-register .js scripts execute on startup and add an icon button on the horizontal Tool Shelf Dock below the menu bar.
~/.config/minicut/scripts/ Menu Click / On Demand .js scripts appear in the top-level Scripts menu on the main menu bar.
~/.config/minicut/icons/ Dock Icon Asset 32x32 PNG icon images matching script stems (e.g., text_animator.png for text_animator.js).

2. JavaScript Plugin Architecture

// ============================================================================
// Example JavaScript Plugin: custom_animator.js
// ============================================================================

var targetTrack = 0;
var textContent = "日本語テキスト字幕";

function applyAnimation() {
    minicut.push_undo();
    var clipId = timeline.place_text_clip(targetTrack, textContent, timeline.get_playhead(), 5.0);
    if (clipId) {
        clip.set_position(clipId, 0.0, 150.0);
        clip.set_scale(clipId, 1.2);
        timeline.add_keyframe(clipId, "opacity", 0.0, 0.0);
        timeline.add_keyframe(clipId, "opacity", 0.5, 1.0);
        minicut.set_status("Placed text clip: " + textContent);
    }
}

plugin.init_panel = function() {
    gui.text("JavaScript Text Animator");
    gui.separator();
    targetTrack = gui.slider_int("Track Lane", targetTrack, 0, 4);
    textContent = gui.input_text("Caption Text", textContent);
    
    if (gui.button("Place Animated Text", applyAnimation)) {
        // inline logic
    }
};
