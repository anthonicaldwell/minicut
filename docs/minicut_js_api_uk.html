1. Вступ та JavaScript архітектура

minicut містить вбудований рушій JavaScript (Duktape 2.7.0) із повноцінною підтримкою кодування UTF-8. Скрипти JavaScript мають повну функціональну еквівалентність із Lua API для розробки потужних локалізованих інструментів.

Каталоги розташування скриптів

РозташуванняРольОпис
~/.config/minicut/autorun/Автозапуск при стартіСкрипти .js виконуються під час запуску та додають кнопку на панель Доку.
~/.config/minicut/scripts/Запуск з менюСкрипти .js відображаються в меню Скрипти.
~/.config/minicut/icons/Іконки ДокуPNG файли іконок, що відповідають іменам скриптів.

2. Архітектура JavaScript плагіна

var targetTrack = 0;
var clipDuration = 4.0;
var mediaPath = "test_sample.png";

function placeMedia() {
    minicut.push_undo();
    var playhead = timeline.get_playhead();
    var cid = timeline.place_clip(mediaPath, targetTrack, playhead, clipDuration);
    if (cid && cid.length > 0) {
        clip.set_scale(cid, 1.25);
        timeline.add_keyframe(cid, "opacity", 0.0, 0.0);
        timeline.add_keyframe(cid, "opacity", 1.0, 1.0);
        minicut.set_status("Кліп розміщено: " + cid);
    }
}

function browseFile() {
    gui.file_dialog("Оберіть файл (JS)", function(selected) {
        if (selected && selected.length > 0) {
            mediaPath = selected;
            minicut.set_status("Обрано: " + selected);
        }
    });
}

plugin.init_panel = function() {
    gui.text_colored(0.24, 0.68, 0.91, 1.0, "Інструмент розміщення медіа (JS)");
    gui.separator();
    targetTrack = gui.slider_int("Номер доріжки", targetTrack, 0, 4);
    clipDuration = gui.slider_float("Тривалість (с)", clipDuration, 0.5, 30.0);
    gui.button("Огляд файлів...", browseFile);
    gui.button("Розмістити на таймлайні", placeMedia);
};

3. Модулі API

У JavaScript доступні всі основні модулі: project, timeline, clip, gui, net, minicut.
